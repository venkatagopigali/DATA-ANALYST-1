constraints :-- will send the data into table based 		condition or will send the data after 		validation

unique            :-- will not accepts duplicate but 			accepts null value
not null          :-- will not accepts the null value
		     will accepts the duplicates values
check             :-- will send the data into column 			after condition is validated
default
auto_increment

column_name datatype constraint
);

create table table_name(
column_name datatype check (condition)
);
